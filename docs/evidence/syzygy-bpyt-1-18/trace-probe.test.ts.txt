const encoder = new TextEncoder();

function sha1Blob(bytes: Uint8Array): string {
  return createHash('sha1').update(`blob ${bytes.byteLength}\0`).update(bytes).digest('hex');
}

function sha256(text: string): string {
  return `sha256:${createHash('sha256').update(text).digest('hex')}`;
}

interface Fixture {
  readonly texts: Readonly<Record<string, string>>;
  readonly failRevParse?: boolean;
  readonly failPhaseBLsTree?: boolean;
  readonly dropFromTree?: readonly string[];
}

function fixture(input: Fixture) {
  const bytes = new Map(Object.entries(input.texts).map(([path, text]) => [path, encoder.encode(text)]));
  const entries: GitTreeEntry[] = [...bytes.entries()]
    .filter(([path]) => !(input.dropFromTree ?? []).includes(path))
    .map(([path, body]) => ({ mode: '100644', type: 'blob', objectId: sha1Blob(body), sizeBytes: body.byteLength, path }));
  const byOid = new Map(entries.map((entry) => [entry.objectId, bytes.get(entry.path) as Uint8Array]));
  const listing = entries.map((e) => `${e.mode} ${e.type} ${e.objectId} ${String(e.sizeBytes).padStart(7)}\t${e.path}\x00`).join('');
  const calls: string[][] = [];
  let lsTreeCalls = 0;
  const runGit = (args: readonly string[]): Uint8Array => {
    calls.push([...args]);
    switch (args[0]) {
      case 'rev-parse':
        if (input.failRevParse === true) throw new Error('fatal: Needed a single revision');
        return encoder.encode(`${COMMIT}\n`);
      case 'show':
        return encoder.encode(`${COMMITTED_AT}\n`);
      case 'ls-tree':
        lsTreeCalls += 1;
        if (args[4] !== COMMIT) throw new Error(`fixture lists trees only at the resolved commit, not ${args[4] ?? ''}`);
        if (input.failPhaseBLsTree === true && lsTreeCalls === 2) throw new Error('fatal: not a tree object');
        return encoder.encode(listing);
      case 'cat-file': {
        const body = byOid.get(args[2] ?? '');
        if (body === undefined) throw new Error(`fixture has no object ${args[2] ?? ''}`);
        return body;
      }
      default:
        throw new Error(`unexpected git command ${args.join(' ')}`);
    }
  };
  return { runGit, calls, entries };
}

function build(input: Fixture & { authority?: BodyReadAuthorityEvaluation; revision?: string; limits?: Partial<PwbResourceLimits> }) {
  const git = fixture(input);
  const shape = buildProjectShape({
    authority: input.authority ?? ADMITTING,
    revision: input.revision ?? 'main',
    capturedAt: CAPTURED_AT,
    runGit: git.runGit,
    ...(input.limits === undefined ? {} : { resourceLimits: { ...PWB_RESOURCE_LIMITS, ...input.limits } }),
  });
  return { shape, calls: git.calls, entries: git.entries };
}

function observed(input: Parameters<typeof build>[0]): Extract<ProjectShape, { kind: 'observed' }> {
  const { shape } = build(input);
  if (shape.kind !== 'observed') throw new Error(`expected observed, got ${shape.kind}`);
  return shape;
}

import { writeFileSync } from 'node:fs';

const TERMS: Record<string, string> = { LF: '\n', CRLF: '\r\n', CR: '\r' };
const fenceText = (n: number, term: string): string => ['```', ...Array.from({ length: n }, () => 'x'), '```', ''].join(term);
const attempt = (fn: () => unknown): string => { try { const r = fn() as { kind?: string }; return `ok:${r.kind ?? typeof r}`; } catch (e) { return `throw:${(e as Error).name}:${(e as Error).message}`; } };

describe('step 1 probe', () => {
  it('measures', async () => {
    const out: Record<string, unknown> = { node: process.version };
    // A. thresholds: the largest closed-fence line count the mask survives, per terminator
    const threshold: Record<string, unknown> = {};
    for (const [name, term] of Object.entries(TERMS)) {
      let lo = 1000, hi = 400000; // lo survives, hi throws
      if (!attempt(() => maskMarkdownCodeContexts(fenceText(hi, term))).startsWith('throw')) { threshold[name] = 'no throw at 400000'; continue; }
      while (hi - lo > 1) { const mid = (lo + hi) >> 1; if (attempt(() => maskMarkdownCodeContexts(fenceText(mid, term))).startsWith('throw')) hi = mid; else lo = mid; }
      threshold[name] = { largestSurviving: lo, firstThrowing: hi, bytesAtFirstThrow: fenceText(hi, term).length };
    }
    out.maskThreshold = threshold;
    // B. the scanner and call sites over 150,000 / 200,000 lines
    const direct: Record<string, unknown> = {};
    for (const n of [100000, 150000, 200000]) for (const [name, term] of Object.entries(TERMS)) {
      const t = fenceText(n, term);
      direct[`${name}-${n}`] = { bytes: t.length, mask: attempt(() => maskMarkdownCodeContexts(t)), scanActiveContent: attempt(() => scanActiveContent(t)) };
    }
    out.direct = direct;
    // C. buildProjectShape: the long fence in a phase-B class source (vision.md) and in the phase-A root index
    const shapes: Record<string, unknown> = {};
    for (const [where, path] of [['phase-B vision.md', 'about/heart-and-soul/vision.md'], ['phase-A root index', 'about/README.md']] as const) {
      for (const n of [100000, 200000]) for (const [name, term] of Object.entries(TERMS)) {
        const base = BASE_TEXTS[path] as string;
        const texts = { ...BASE_TEXTS, [path]: `${base}\n${fenceText(n, term)}` };
        const git = fixture({ texts });
        let result = '';
        let detail: unknown;
        try {
          const shape = buildProjectShape({ authority: ADMITTING, revision: 'main', capturedAt: CAPTURED_AT, runGit: git.runGit });
          result = `returned:${shape.kind}`;
          if (shape.kind === 'observed') { const s = shape.sources.find((x) => x.path === path); detail = { sourceRecord: s?.record.outcome, exclusion: s?.record.outcome === 'excluded' ? s.record.exclusion : undefined, wholeShape: shape.claim.label }; }
        } catch (e) { result = `threw:${(e as Error).name}:${(e as Error).message}`; }
        shapes[`${where} ${name} ${n} lines (${texts[path].length} bytes)`] = { result, detail, gitCatFileCalls: git.calls.filter((c) => c[0] === 'cat-file').length };
      }
    }
    out.buildProjectShape = shapes;
    // D. re-observation: the state wrapper around a build that hits the long fence
    const longTexts = { ...BASE_TEXTS, 'about/heart-and-soul/vision.md': `${VISION}\n${fenceText(200000, '\n')}` };
    const okGit = fixture({ texts: BASE_TEXTS });
    const goodModel = buildProjectShape({ authority: ADMITTING, revision: 'main', capturedAt: CAPTURED_AT, runGit: okGit.runGit });
    const state = createReobserveState<string, ProjectShape>({
      capture: 'c0', model: goodModel, observe: async () => 'c1',
      build: () => buildProjectShape({ authority: ADMITTING, revision: 'main', capturedAt: CAPTURED_AT, runGit: fixture({ texts: longTexts }).runGit }),
    });
    const res = await state.reobserve();
    out.reobserveState = { result: res.kind, reason: res.kind === 'failed' ? res.reason : undefined, priorModelStillServed: state.get() === goodModel, captureStillC0: state.getCapture() === 'c0' };
    const routes = reobserveRoutes({ reobserve: async () => ({ kind: 'failed', reason: res.kind === 'failed' ? res.reason : '' }) });
    const human = routes.find((r) => r.method === 'POST');
    out.reobserveRouteCount = routes.length;
    out.reobserveRouteHuman = human === undefined ? 'none' : { path: human.path, credentialClass: human.credentialClass };
    // E. startup
    out.startupStderrLine = `syzygy POC: observation failed (${unexpectedObservationDetail(new RangeError('Maximum call stack size exceeded'))})`;
    writeFileSync(`/tmp/lci/trace-${process.env['PROBE_TAG']}-1.json`, JSON.stringify(out, null, 1));
    expect(out).toBeDefined();
  }, 600000);
});
