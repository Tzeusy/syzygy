const HOME = { headers: { host: '127.0.0.1:7478', origin: 'http://127.0.0.1:7478' } };
const reqCtx = (method: string, path: string) => ({ request: { method, path, query: new URLSearchParams(), headers: HOME.headers }, admission: undefined });

async function hit(routes: Route[], method: string, path: string): Promise<string> {
  const route = routes.find((r) => r.method === method && r.path === path);
  if (route === undefined) return `${method} ${path}: no route`;
  const response = await route.handle(reqCtx(method, path));
  return `${method} ${path}: ${response.status} ${response.contentType} len=${response.body.length} sha=${createHash('sha256').update(response.body).digest('hex')}`;
}

interface Transcript { stdout: string; stderr: string; exitCode: number; calls: string[]; thrown: string | null; after: string[]; closed: number }

function normalize(t: Transcript): Transcript { return t; }

async function runScript(scenario: Scenario, load: () => Promise<unknown>): Promise<Transcript> {
  fx.calls = []; fx.scenario = scenario; fx.repoN = 0; fx.cleanN = 0; fx.horizonN = 0; fx.modelN = 0; fx.daemonOptions = null; fx.afterDone = []; fx.closed = 0;
  let stdout = ''; let stderr = '';
  fx.stdin = new PassThrough();
  const stdinDescriptor = Object.getOwnPropertyDescriptor(process, 'stdin')!;
  Object.defineProperty(process, 'stdin', { value: fx.stdin, configurable: true });
  const savedArgv = process.argv;
  process.argv = ['node', '/x/dist/main.js', ...scenario.argv];
  process.exitCode = undefined;
  const out = vi.spyOn(process.stdout, 'write').mockImplementation(((chunk: unknown) => { stdout += String(chunk); return true; }) as never);
  const err = vi.spyOn(process.stderr, 'write').mockImplementation(((chunk: unknown) => { stderr += String(chunk); return true; }) as never);
  const cwd = vi.spyOn(process, 'cwd').mockReturnValue(scenario.cwd ?? '/observer/syzygy');
  const before = { SIGINT: process.listeners('SIGINT'), SIGTERM: process.listeners('SIGTERM') };
  fx.emit = (signal) => { process.emit(signal, signal); };
  vi.resetModules();
  // The error class identity belongs to the module registry under test.
  fx.PocError = (await import('@syzygy/three-surface-poc-core')).PocObservationError;
  let thrown: string | null = null;
  try {
    await load();
  } catch (cause) {
    thrown = cause instanceof Error ? `${cause.name}: ${cause.message}` : String(cause);
  } finally {
    for (const signal of ['SIGINT', 'SIGTERM'] as const) for (const listener of process.listeners(signal)) if (!before[signal].includes(listener)) process.removeListener(signal, listener);
    out.mockRestore(); err.mockRestore(); cwd.mockRestore();
    Object.defineProperty(process, 'stdin', stdinDescriptor);
  }
  const exitCode = typeof process.exitCode === 'number' ? process.exitCode : 0;
  process.exitCode = undefined;
  process.argv = savedArgv;
  return normalize({ stdout, stderr, exitCode, calls: fx.calls, thrown, after: fx.afterDone, closed: fx.closed });
}

const runOld = (scenario: Scenario) => runScript(scenario, () => import('./zz-main-old.js'));

async function runNew(scenario: Scenario): Promise<Transcript> {
  fx.calls = []; fx.scenario = scenario; fx.repoN = 0; fx.cleanN = 0; fx.horizonN = 0; fx.modelN = 0; fx.daemonOptions = null; fx.afterDone = []; fx.closed = 0;
  fx.stdin = new PassThrough();
  let stdout = ''; let stderr = '';
  const collect = (sink: (text: string) => void) => new Writable({ write(chunk, _encoding, callback) { sink(String(chunk)); callback(); } });
  const listeners: { signal: 'SIGINT' | 'SIGTERM'; listener: () => void }[] = [];
  fx.emit = (signal) => { for (const l of listeners.filter((entry) => entry.signal === signal)) l.listener(); };
  const cwdSpy = vi.spyOn(process, 'cwd').mockReturnValue(scenario.cwd ?? '/observer/syzygy');
  const { main } = await import('./poc-main.js');
  let thrown: string | null = null;
  let exitCode = -1;
  try {
    exitCode = await main(scenario.argv, {
      stdout: collect((t) => { stdout += t; }),
      stderr: collect((t) => { stderr += t; }),
      stdin: fx.stdin,
      cwd: () => scenario.cwd ?? '/observer/syzygy',
      now: () => new Date().toISOString(),
      onSignal: (signal, listener) => { listeners.push({ signal, listener }); },
    });
  } catch (cause) {
    thrown = cause instanceof Error ? `${cause.name}: ${cause.message}` : String(cause);
    exitCode = 0;
  }
  cwdSpy.mockRestore();
  return normalize({ stdout, stderr, exitCode, calls: fx.calls, thrown, after: fx.afterDone, closed: fx.closed });
}
